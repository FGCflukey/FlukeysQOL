   color mask rather than needing a new sprite, tinted per group to

--------------------------------------------------------------- */

/* ---------------------------------------------------------------
   Pack / Unpack recipes -- structured exactly like vanilla's own
   PackSetOfBooks / UnpackSetOfBooks (generated/recipes/
   recipes_packing.txt), just scoped to the 9 skill lines above
   instead of vanilla's skill books. One shared Pack recipe (5 input
   slots, one per level, mapped through a single itemMapper) and one
   shared Unpack recipe (1 input slot accepting any of the 9 Set
   items, mapped through 5 separate itemMapper blocks, one per
   output level) -- same shape vanilla uses so all 9 skill lines
   ride through two recipes instead of eighteen.
--------------------------------------------------------------- */

    craftRecipe PackAdditionalBookSet
    {
        timedAction = PackingBox,
        time        = 80,
        Tags        = InHandCraft;Packing,
        category    = Packing,
        inputs
        {
            item 1 [Base.BookAxe1;Base.BookBlunt1;Base.BookSmallBlunt1;Base.BookSmallBlade1;Base.BookSpear1;Base.BookSprinting1;Base.BookNimble1;Base.BookSneaking1;Base.BookLightfooted1] mappers[bookType] flags[AllowFavorite;InheritFavorite],
            item 1 [Base.BookAxe2;Base.BookBlunt2;Base.BookSmallBlunt2;Base.BookSmallBlade2;Base.BookSpear2;Base.BookSprinting2;Base.BookNimble2;Base.BookSneaking2;Base.BookLightfooted2] mappers[bookType] flags[AllowFavorite;InheritFavorite],
            item 1 [Base.BookAxe3;Base.BookBlunt3;Base.BookSmallBlunt3;Base.BookSmallBlade3;Base.BookSpear3;Base.BookSprinting3;Base.BookNimble3;Base.BookSneaking3;Base.BookLightfooted3] mappers[bookType] flags[AllowFavorite;InheritFavorite],
            item 1 [Base.BookAxe4;Base.BookBlunt4;Base.BookSmallBlunt4;Base.BookSmallBlade4;Base.BookSpear4;Base.BookSprinting4;Base.BookNimble4;Base.BookSneaking4;Base.BookLightfooted4] mappers[bookType] flags[AllowFavorite;InheritFavorite],
            item 1 [Base.BookAxe5;Base.BookBlunt5;Base.BookSmallBlunt5;Base.BookSmallBlade5;Base.BookSpear5;Base.BookSprinting5;Base.BookNimble5;Base.BookSneaking5;Base.BookLightfooted5] mappers[bookType] flags[AllowFavorite;InheritFavorite],
        }
        outputs
        {
            item 1 mapper:bookType,
        }
        itemMapper bookType
        {
            Base.BookAxeSet         = Base.BookAxe1;Base.BookAxe2;Base.BookAxe3;Base.BookAxe4;Base.BookAxe5,
            Base.BookBluntSet       = Base.BookBlunt1;Base.BookBlunt2;Base.BookBlunt3;Base.BookBlunt4;Base.BookBlunt5,
            Base.BookSmallBluntSet  = Base.BookSmallBlunt1;Base.BookSmallBlunt2;Base.BookSmallBlunt3;Base.BookSmallBlunt4;Base.BookSmallBlunt5,
            Base.BookSmallBladeSet  = Base.BookSmallBlade1;Base.BookSmallBlade2;Base.BookSmallBlade3;Base.BookSmallBlade4;Base.BookSmallBlade5,
            Base.BookSpearSet       = Base.BookSpear1;Base.BookSpear2;Base.BookSpear3;Base.BookSpear4;Base.BookSpear5,
            Base.BookSprintingSet   = Base.BookSprinting1;Base.BookSprinting2;Base.BookSprinting3;Base.BookSprinting4;Base.BookSprinting5,
            Base.BookNimbleSet      = Base.BookNimble1;Base.BookNimble2;Base.BookNimble3;Base.BookNimble4;Base.BookNimble5,
            Base.BookSneakingSet    = Base.BookSneaking1;Base.BookSneaking2;Base.BookSneaking3;Base.BookSneaking4;Base.BookSneaking5,
            Base.BookLightfootedSet = Base.BookLightfooted1;Base.BookLightfooted2;Base.BookLightfooted3;Base.BookLightfooted4;Base.BookLightfooted5,
        }
    }
